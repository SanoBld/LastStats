static Future<String?> generate(String system, String user,
    {int maxTokens = 140}) async {
  final m = await activeModel();
  if (m == null) return null;
  final path = (await _file(m)).path;
  try {
    final out = await _enqueue(() => _run(path, system, user, maxTokens));
    final t = out.trim();
    lastError = t.isEmpty ? 'empty output' : '';
    return t.isEmpty ? null : t;
  } catch (e) {
    lastError = e.toString();
    return null;
  }
}
