static Future<List<HfFile>> listFiles(String repo) async {
  final uri = Uri.https(_hf, '/api/models/$repo/tree/main');
  final r = await http.get(uri).timeout(const Duration(seconds: 15));
  if (r.statusCode != 200) throw HttpException('HTTP ${r.statusCode}');
  final out = <HfFile>[];
  for (final e in jsonDecode(r.body) as List) {
    final path = (e['path'] ?? '').toString();
    final low = path.toLowerCase();
    if (e['type'] != 'file' || !low.endsWith('.gguf')) continue;
    if (low.contains('mmproj') || RegExp(r'-\d{5}-of-\d{5}').hasMatch(low)) continue;
    final size = (e['lfs']?['size'] ?? e['size'] ?? 0) as int;
    out.add(HfFile(path, (size / 1048576).round()));
  }
  out.sort((a, b) => a.sizeMb.compareTo(b.sizeMb));
  return out;
}
